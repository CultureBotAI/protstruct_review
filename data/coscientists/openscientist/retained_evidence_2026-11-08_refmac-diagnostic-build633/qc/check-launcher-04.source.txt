# Actual main must refuse mismatched role/request identities before its claim.
# All product boundaries are mocked; the native collector is never entered.
def main_role_case(name, req, mutate=None, executable=None, expect_error=False):
    request_path.write_text(json.dumps(req))
    local_argv = [str(source), '--request', str(request_path), '--request-sha256', m.digest(request_path),
                  '--request-size-bytes', str(request_path.stat().st_size)]
    prov = valid_provenance(req)
    if mutate:
        mutate(prov)
    local_dp = SimpleNamespace(**vars(fake_dp))
    local_dp.validate_profile = lambda *a: Baseline(prov)
    previous_path = list(sys.path)
    try:
        with patch.object(sys, 'argv', local_argv), patch.object(sys, 'executable', executable or sys.executable), \
             patch.object(m, 'checked_pin', fake_pincheck), patch.object(m, 'safe_fresh_paths'), \
             patch.object(m, 'read_git_state', return_value=git_baseline), \
             patch.object(m, 'run_claimed_once', return_value={'collector_returned': False}) as claim, \
             patch.dict(os.environ, {'TMUX_PANE':'%0', 'HOME':'/synthetic', 'PATH':'/untrusted'}, clear=True), \
             patch.dict(sys.modules, {'refmac_diagnostic_profile':local_dp, 'onesar_refmac':fake_col, 'onesar_rerun':fake_rerun}), \
             redirect_stdout(io.StringIO()):
            if expect_error:
                try:
                    m.main()
                except ValueError as exc:
                    assert 'Execution role mismatch:' in str(exc)
                else:
                    raise AssertionError(name + ' accepted')
                claim.assert_not_called()
            else:
                m.main()
                claim.assert_called_once()
                retained = claim.call_args.args[0]['execution_role_joins']
                assert set(retained) == {'launcher', 'interpreter'}
                for key in retained:
                    assert retained[key]['matches'] is True
                    assert retained[key]['request_observation'] == retained[key]['profile_observation']
    finally:
        sys.path[:] = previous_path
    checks.append(name)

main_role_case('matched_actual_main_role_join', copy.deepcopy(request))
