def pin(path: Path) -> dict:
    if path.is_symlink() or not path.is_file():
        raise ValueError(f"Expected regular non-symlink file: {path}")
    if path.stat().st_size > 1024 * 1024 * 1024:
        raise ValueError(f"Input exceeds 1 GiB control limit: {path}")
    digest = hashlib.sha256()
    size = 0
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
            size += len(chunk)
            if size > 1024 * 1024 * 1024:
                raise ValueError(f"Input grew beyond 1 GiB control limit: {path}")
    return {"sha256": digest.hexdigest(), "size_bytes": size}


def dump(path: Path, payload: object) -> None:
    path.write_text(json.dumps(payload, indent=2, sort_keys=True, allow_nan=False) + "\n")


def exception_record(exc: BaseException) -> dict:
    return {
        "type": type(exc).__name__, "message": str(exc),
        "traceback": "".join(traceback.format_exception(type(exc), exc, exc.__traceback__)),
    }


class DeniedActivity(RuntimeError):
    pass


def install_guards(output: Path, denied: list) -> None:
    """Install before any candidate import; latch denials even if product catches."""
    def deny(label):
        def blocked(*args, **kwargs):
            denied.append({"event": label})
            raise DeniedActivity(f"Forbidden process/network/native operation: {label}")
        return blocked

    def owned(raw):
        if isinstance(raw, int):
            return
        path = Path(os.fsdecode(raw)).absolute().resolve(strict=False)
        if path != output and not path.is_relative_to(output):
            denied.append({"event": "write_outside_owned_output", "path": str(path)})
            raise DeniedActivity(f"Write outside owned output: {path}")

    def audit(event, args):
        if (event.startswith("subprocess.") or event.startswith("socket.")
                or event.startswith("ctypes.")
                or event in {"os.system", "os.exec", "os.posix_spawn", "os.fork",
                             "os.forkpty", "os.startfile", "os.startfile/2"}):
            deny(event)()
        if event == "open":
            raw, mode, flags = args
            if ((isinstance(mode, str) and any(x in mode for x in "wax+"))
                    or (isinstance(flags, int) and flags & (
                        os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND))):
                owned(raw)
        elif event in {"os.mkdir", "os.remove", "os.rmdir", "os.chmod",
                       "os.chown", "os.utime", "os.truncate"}:
            owned(args[0])
        elif event in {"os.rename", "os.link"}:
            owned(args[0])
            owned(args[1])
        elif event == "os.symlink":
            owned(args[1])

    sys.addaudithook(audit)
    for name in ("Popen", "run", "call", "check_call", "check_output",
                 "getoutput", "getstatusoutput"):
        setattr(subprocess, name, deny("subprocess." + name))
    for name in dir(os):
        if name.startswith(("exec", "spawn", "posix_spawn")) or name in {
            "system", "fork", "forkpty", "startfile", "popen", "_exit"
        }:
            if callable(getattr(os, name)):
                setattr(os, name, deny("os." + name))
    for name in ("socket", "socketpair", "create_connection", "create_server",
                 "fromfd", "getaddrinfo", "gethostbyname", "gethostbyname_ex"):
        if hasattr(socket, name):
            setattr(socket, name, deny("socket." + name))
    for name in ("CDLL", "PyDLL", "WinDLL", "OleDLL"):
        if hasattr(ctypes, name):
            setattr(ctypes, name, deny("ctypes." + name))
    for name in ("cdll", "pydll", "windll", "oledll"):
        if hasattr(ctypes, name):
            getattr(ctypes, name).LoadLibrary = deny("ctypes." + name + ".LoadLibrary")


def source_inventory(scripts: Path) -> dict:
    rows = {}
    total = 0
    for path in sorted(scripts.rglob("*")):
        if path.is_symlink():
            raise ValueError(f"Symlink in candidate source tree: {path}")
        if path.is_file() and path.suffix == ".py":
            if path.stat().st_size > 5 * 1024 * 1024 or len(rows) >= 1024:
                raise ValueError("Source inventory exceeds bounded control limits")
            row = pin(path)
            total += row["size_bytes"]
            if row["size_bytes"] > 5 * 1024 * 1024 or total > 64 * 1024 * 1024:
                raise ValueError("Source inventory exceeds bounded control limits")
            rows[path.relative_to(scripts).as_posix()] = row
    if not rows or len(rows) > 1024:
        raise ValueError("Expected 1..1024 pinned Python source files")
    return rows


class PinnedSourceFinder:
    """Use verified source bytes, never an existing candidate bytecode cache."""
    def __init__(self, scripts, pins, imported):
        self.scripts, self.pins, self.imported = scripts, pins, imported

    def find_spec(self, fullname, path=None, target=None):
        spec = importlib.machinery.PathFinder.find_spec(fullname, path)
        if spec is None or not spec.origin or spec.origin in {"built-in", "frozen"}:
            return None
        origin = Path(spec.origin).resolve(strict=True)
        if not origin.is_relative_to(self.scripts):
            if fullname.replace(".", "/") + ".py" in self.pins:
                raise ImportError(f"Candidate module resolved outside pinned scripts: {fullname}: {origin}")
            return None
        relative = origin.relative_to(self.scripts).as_posix()
        expected = self.pins.get(relative)
        if expected is None or origin.suffix != ".py" or pin(origin) != expected:
            raise ImportError(f"Unpinned candidate import: {fullname}: {origin}")
        raw = origin.read_bytes()
        if {"sha256": hashlib.sha256(raw).hexdigest(), "size_bytes": len(raw)} != expected:
            raise ImportError(f"Candidate source changed while loading: {origin}")
        self.imported[fullname] = {"path": str(origin), **expected}

        class Loader(importlib.machinery.SourceFileLoader):
            def get_code(self, module_name):
                return self.source_to_code(raw, str(origin))

        return importlib.util.spec_from_file_location(
            fullname, str(origin), loader=Loader(fullname, str(origin)),
            submodule_search_locations=spec.submodule_search_locations,
        )


class Case:
    def __init__(self, name, root):
        self.name, self.root = name, root
        root.mkdir()
        self.record = {"name": name, "status": "running", "checks": [], "events": []}

    def check(self, condition, label, detail=None):
        self.record["checks"].append({"passed": bool(condition), "assertion": label,
                                      "detail": detail})

    def event(self, name):
        self.record["events"].append(name)


def reopen_reference(reference, output):
    path = Path(reference["path"])
    resolved = path.resolve(strict=True)
    if path != resolved or not resolved.is_relative_to(output):
        raise ValueError(f"Unexpected retained reference destination: {path}")
    actual = pin(path)
    if any(actual[k] != reference[k] for k in ("sha256", "size_bytes")):
        raise AssertionError(f"Retained reference pins mismatch: {path}")
    return json.loads(path.read_text())

def retained_pair(case, error, output):
    if error.manifest_path is None or error.diagnostics_path is None:
        raise AssertionError("Expected persisted authoritative manifest and diagnostics")
    manifest_path = Path(error.manifest_path)
    manifest_ref = {"path": str(manifest_path), **pin(manifest_path)}
    manifest = reopen_reference(manifest_ref, output)
    diagnostics = reopen_reference(manifest["diagnostics_json"], output)
    case.check(manifest == error.manifest_payload, "Persisted manifest equals actual error payload")
    case.check(diagnostics == error.diagnostics_payload, "Persisted diagnostics equal actual error payload")
    case.check(error.publication_pair_complete is True, "Publication pair complete")
    case.check(str(error.diagnostics_path) == manifest["diagnostics_json"]["path"],
               "Error diagnostics path binds pinned manifest reference")
    case.record["manifest_reference"] = manifest_ref
    case.record["diagnostics_reference"] = manifest["diagnostics_json"]
    return manifest, diagnostics

        else:

                       and envelope["first_failure"]["message"] == "ValueError: " + str(trace_error),
                       "Actual trace inspection exception retained")
        else:
            actual = case.record["actual_trace_result"]
            case.check(envelope["valid"] is False and envelope["errors"] == actual["errors"],
                       "Actual malformed-trace findings retained")
    for key in ("pre_execution_checkpoint_json", "post_execution_checkpoint_json"):
        document = reopen_reference(manifest[key], output)
        case.check(type(document["schema_version"]) is int and document["schema_version"] == 2

    case.check(manifest["toolchain"]["refmac5_binary"]["resolved_path"] == str(fixture.installed_refmac)
               and str(fixture.installed_refmac) != str(fixture.diagnostic_executable_path),
               "Installed identity remains distinct from selected diagnostic executable")
    if mode == "combined":
        case.check(caught.failed_stage == "refmac_runner_invocation" and
                   caught.primary_cause is runner_error and caught.__cause__ is runner_error,
                   "Original runner exception remains primary and explicit cause")
        case.check(any(item is profile_error for item in lists["mutations"]),
                   "Original profile exception retained in aligned mutation-cause list")
        case.check(any(item is profile_error for item in lists["inspection"]) and
                   any(item is trace_error for item in lists["inspection"]),
                   "Both original later inspection objects retained")
        for document in (manifest, diagnostics):
            stages = [row["failed_stage"] for row in document["failures"]]
            case.check(stages[0] == "refmac_runner_invocation" and
                       "verify_input_immutability" in stages and "refmac_raw_outputs" in stages,
                       "Persisted failures retain primary and both later channels", stages)
    elif mode == "invocation":
        case.check(caught.failed_stage == "refmac_raw_outputs",
                   "Invocation case fails specifically on deliberately invalid trace", caught.failed_stage)


def run_consumer(case, modules, output):
    records = modules[4]
    repo = case.root / "repo"
    repo.mkdir()
    for schema, fragment in ((2, "expected integer schema_version == 1"),
                             (1, "refmac_operation_manifest.collector_stage: expected")):
        # Bound only to the inspected early gates; no valid-science fixture is claimed.
        document = dict.fromkeys(CONSUMER_KEYS)
        document.update(schema_version=schema, collector_stage=STAGE, subject_id="R4")
        path = case.root / f"schema-{schema}.json"
        dump(path, document)
        reference = {"path": str(path), **pin(path)}
        try:
            records.verify_retained_refmac_rfactor_evidence(
                case.root, subject_id="R4", manifest_reference=reference, repo_root=repo,
            )
        except records.RefmacRetainedVerificationError as error:
            case.record[f"schema_{schema}_error"] = exception_record(error)
            case.check(fragment in str(error), "Specific legacy refusal for schema " + str(schema), str(error))
            if schema == 1:
                case.check(LEGACY_STAGE in str(error) and STAGE in str(error),
                           "Stage refusal identifies both literal stages")
        else:
            case.check(False, "Legacy reader must reject schema/stage " + str(schema))
        case.check(reopen_reference(reference, output) == document, "Consumer input unchanged and hash-consistent")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--scripts", type=Path, required=True)
    parser.add_argument("--pins", type=Path, required=True,
                        help="JSON mapping EVERY scripts/**/*.py relative name to sha256,size_bytes")
    parser.add_argument("--collector-sha256", required=True)
    parser.add_argument("--fixture-sha256", required=True)
    parser.add_argument("--archive", type=Path, required=True)
    parser.add_argument("--archive-sha256", required=True)
    parser.add_argument("--archive-size", type=int, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    evidence_parent = Path(__file__).resolve(strict=True).parent.parent
    output = args.output.absolute()
    if (evidence_parent.name != "601-instrumented-build" or output.exists() or output.is_symlink()
            or not output.parent.resolve(strict=True).is_relative_to(evidence_parent)
            or output != output.resolve(strict=False)):
        parser.error("--output must be a fresh canonical durable child under 601-instrumented-build")
    output.mkdir()
    result = {"status": "running", "cases": [], "denied_events": [],
              "scope": "C1-C5 bounded synthetic controls; no scientific/native acceptance"}
    before = None
    scripts = None
    archive = None
    imported = {}
    sys.dont_write_bytecode = True
    try:
        install_guards(output, result["denied_events"])
        if sys.flags.optimize:
            raise ValueError("Run without Python optimization")
        scripts = args.scripts.resolve(strict=True)
        if scripts.is_relative_to(output) or output.is_relative_to(scripts):
            raise ValueError("Candidate source tree must be separate from owned output")
        pins = json.loads(args.pins.read_text())
        before = source_inventory(scripts)
        if before != pins:
            raise ValueError("Complete candidate Python-source inventory does not match supplied pins")
        for name, expected in (("onesar_refmac.py", args.collector_sha256),
                               ("test_onesar_refmac_diagnostic.py", args.fixture_sha256)):
            if not re.fullmatch(r"[0-9a-f]{64}", expected) or pins[name]["sha256"] != expected:
                raise ValueError(f"Explicit final source pin mismatch: {name}")
        dump(output / "source-before.json", before)
        archive = args.archive.resolve(strict=True)
        if (args.archive_sha256 != ARCHIVE_SHA or args.archive_size <= 0
                or args.archive_size > 1024 * 1024 * 1024
                or archive.stat().st_size != args.archive_size or pin(archive) != {
                    "sha256": args.archive_sha256, "size_bytes": args.archive_size}):
            raise ValueError("PRIMARY-supplied actual committed archive pin mismatch")
        result["archive"] = {"path": str(archive), **pin(archive)}
        result["pins_document"] = {"path": str(args.pins.resolve()), **pin(args.pins)}
        for name in (relative[:-3].replace("/", ".") for relative in pins):
            if name in sys.modules:
                raise ValueError(f"Candidate was imported before guards: {name}")
        sys.path.insert(0, str(scripts))
        sys.meta_path.insert(0, PinnedSourceFinder(scripts, pins, imported))
        modules = tuple(importlib.import_module(name) for name in (
            "test_onesar_refmac_diagnostic", "onesar_refmac", "refmac_diagnostic_profile",
            "refmac_diagnostic_trace", "onesar_records",
        ))
        cases = [
            ("c1-helper-target", lambda case: run_helper_target(case, modules, archive, output)),
            ("c2-one-invocation", lambda case: run_capture(case, modules, archive, output, "invocation")),
            ("c3a-profile-error", lambda case: run_capture(case, modules, archive, output, "profile_exception")),
            ("c3b-trace-error", lambda case: run_capture(case, modules, archive, output, "trace_exception")),
            ("c4-combined-errors", lambda case: run_capture(case, modules, archive, output, "combined")),
            ("c5-consumer", lambda case: run_consumer(case, modules, output)),
        ]
        for name, function in cases:
            case = Case(name, output / name)
            result["cases"].append(case.record)
            try:
                function(case)
            except BaseException as error:
                case.record["unexpected_error"] = exception_record(error)
            case.record["status"] = (
                "passed" if case.record["checks"] and "unexpected_error" not in case.record
                and all(row["passed"] for row in case.record["checks"]) else "failed"
            )
            dump(case.root / "control-result.json", case.record)
            dump(output / "results.json", result)
    except BaseException as error:
        result["setup_or_execution_error"] = exception_record(error)
    finally:
        if archive is not None and "archive" in result:
            try:
                result["archive_unchanged"] = pin(archive) == {
                    "sha256": result["archive"]["sha256"],
                    "size_bytes": result["archive"]["size_bytes"],
                }
            except BaseException as error:
                result["archive_recheck_error"] = exception_record(error)
        if before is not None and scripts is not None:
            try:
                after = source_inventory(scripts)
                dump(output / "source-after.json", after)
                result["source_inventory_unchanged"] = before == after
                result["imported_candidate_sources"] = imported
                result["all_candidate_imports_pinned"] = all(
                    row["path"] == str(scripts / name) and pin(Path(row["path"])) == before[name]
                    for row in imported.values()
                    for name in [Path(row["path"]).relative_to(scripts).as_posix()]
                )
            except BaseException as error:
                result["source_recheck_error"] = exception_record(error)
        result["status"] = (
            "passed" if len(result["cases"]) == 6
            and all(case["status"] == "passed" for case in result["cases"])
            and not result["denied_events"] and result.get("source_inventory_unchanged") is True
            and result.get("archive_unchanged") is True
            and result.get("all_candidate_imports_pinned") is True
            and "setup_or_execution_error" not in result and "source_recheck_error" not in result
            else "failed"
        )
        dump(output / "results.json", result)
    print(json.dumps({"status": result["status"], "results": str(output / "results.json")}))
    return 0 if result["status"] == "passed" else 1


if __name__ == "__main__":
    raise SystemExit(main())
