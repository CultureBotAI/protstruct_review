def strict_pairs(pairs):
    result = {}
    for k,v in pairs:
        if k in result:
            raise ValueError('Duplicate JSON key '+k)
        result[k]=v
    return result
def read_json(path):
    path=Path(path)
    resolved=path.resolve(strict=True)
    raw=path.read_bytes()
    def reject_constant(token):
        raise ValueError('Nonstandard JSON constant: '+token)
    result=json.loads(raw,object_pairs_hook=strict_pairs,parse_constant=reject_constant)
    register(path,expected={'sha256':hashlib.sha256(raw).hexdigest(),
                           'size_bytes':len(raw),'resolved_path':str(resolved)})
    return result
def digest(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda:f.read(1024*1024),b''):
            h.update(chunk)
    return h.hexdigest()
def fail_walk(exc):
    raise OSError('Incomplete profile tree enumeration: '+str(exc)) from exc
def members(root,exclude_git=False):
    rows={}
    for base,dirs,names in os.walk(root,followlinks=False,onerror=fail_walk):
        if exclude_git and Path(base)==root:
            dirs[:]=[x for x in dirs if x!='.git'];names=[x for x in names if x!='.git']
        for name in dirs+names:
            q=Path(base)/name
            if q.is_symlink():
                raise ValueError('Controlled tree symlink: '+str(q))
        for name in names:
            q=Path(base)/name
            if not stat.S_ISREG(q.stat().st_mode):
                raise ValueError('Nonregular profile member: '+str(q))
            rows[q.relative_to(root).as_posix()]={'sha256':digest(q),'size_bytes':q.stat().st_size}
    return rows
def symlink_edges(path):
    # Preserve aliases, including legitimate bin/../libexec paths. This records
    # literal symlink edges; resolved_path independently binds the final target.
    edges={};pending=[path];seen=set()
    while pending:
        item=pending.pop()
        for q in [item,*item.parents]:
            raw=str(q)
            if raw in seen:
                continue
            seen.add(raw)
            if q.is_symlink():
                target=os.readlink(q);edges[raw]=target
                next_path=Path(target) if Path(target).is_absolute() else q.parent/target
                pending.append(next_path)
    return [{'path':k,'target':v} for k,v in sorted(edges.items())]

def register(path,*,file_id=None,expected=None):
    path=Path(path)
    if not path.is_absolute():
        path=Path.cwd()/path
    raw=str(path);resolved=path.resolve(strict=True)
    st=resolved.stat()
    if not stat.S_ISREG(st.st_mode):
        raise ValueError('Profile file is not regular: '+raw)
    row={'path':raw,'resolved_path':str(resolved),'sha256':digest(path),
         'size_bytes':st.st_size,'symlinks':symlink_edges(path)}
    if expected is not None:
        assert row['sha256']==expected['sha256'],raw
        if 'size_bytes' in expected:
            assert type(expected['size_bytes']) is int and expected['size_bytes']>=0,raw
            assert row['size_bytes']==expected['size_bytes'],raw
        targets=[expected[key] for key in ['resolved','resolved_path'] if key in expected]
        assert all(type(value) is str and value==row['resolved_path'] for value in targets),raw
    if raw in by_path:
        old=by_path[raw]
        assert files[old]==row,raw
        return old
    ident=file_id or 'external:'+hashlib.sha256(raw.encode()).hexdigest()
    assert ident not in files,ident
    by_path[raw]=ident;files[ident]=row
    return ident
def role(name,paths):
    ids=[register(path) for path in paths]
    assert len(set(ids))==len(ids),name
    roles[name]=ids
def exact_role(name,path,expected=None):
    roles[name]=[register(path,expected=expected)]
