def pin(path):
 p=Path(path);h=hashlib.sha256();st=p.lstat()
 if stat.S_ISLNK(st.st_mode):
  target=os.readlink(p);data=os.fsencode(target)
  return {'path':str(p),'kind':'symlink','target':target,'sha256':hashlib.sha256(data).hexdigest(),'size_bytes':len(data),'mode':stat.S_IMODE(st.st_mode)}
 if not stat.S_ISREG(st.st_mode):raise ValueError('Unexpected non-file input: '+str(p))
 with p.open('rb') as f:
  for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
 return {'path':str(p),'sha256':h.hexdigest(),'size_bytes':p.stat().st_size}
