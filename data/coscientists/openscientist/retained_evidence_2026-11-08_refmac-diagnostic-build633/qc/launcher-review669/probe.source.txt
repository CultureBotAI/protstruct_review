def guard(event, args):
    if event in {'subprocess.Popen', 'os.system', 'os.posix_spawn', 'os.fork',
                 'socket.connect', 'socket.bind'}:
        raise AssertionError('Forbidden execution/network: ' + event)
    if event == 'open':
        filename, mode, flags = args
        if isinstance(filename, (str, bytes, os.PathLike)) and flags & (
                os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND):
            assert Path(filename).resolve().is_relative_to(OUT), filename
    if event == 'import' and args[0] in {
            'onesar_refmac', 'onesar_rerun', 'refmac_diagnostic_profile'}:
        raise AssertionError('Real product import forbidden: ' + args[0])

class Validated:
    def __init__(self, payload):
        self.payload = copy.deepcopy(payload)

    def to_provenance(self):
        return copy.deepcopy(self.payload)

def provenance(observations):
    return {'profile_files_verified': True,
            'roles': {'launcher': ['opaque-z9'], 'collection_interpreter': ['opaque-a7']},
            'files': {'opaque-z9': copy.deepcopy(observations['launcher']),
                      'opaque-a7': copy.deepcopy(observations['interpreter'])}}

def expect_refusal(name, fn, message=None):
    try:
        fn()
    except (ValueError, KeyError) as error:
        if message:
            assert message in str(error), str(error)
    else:
        raise AssertionError(name + ' accepted')
    checks.append(name)

        row = changed['files'][file_id]

def request_pin(path):
    return {key: value for key, value in m.pin(path).items() if key != 'resolved_path'}

def main_case(name, mutate=None, reject=False):
    events = []
    actual = {key: m.checked_pin(request[key]) for key in ('launcher', 'interpreter')}
    payload = provenance(actual)
    if mutate:
        mutate(payload)

    def validate(*args):
        events.append('validate_profile')
        return Validated(payload)

    def collector(*args, **kwargs):
        raise AssertionError('Collector must never be entered')

    def claim(start, callable_, profile):
        events.append('claim')
        retained = start['execution_role_joins']
        assert set(retained) == {'launcher', 'interpreter'}
        for key in retained:
            assert retained[key]['request_observation'] == actual[key]
            assert retained[key]['profile_observation'] == actual[key]
            assert retained[key]['matches'] is True
        assert retained['interpreter']['request_observation']['path'] == str(interpreter_alias)
        assert events == ['validate_profile', 'claim']
        return {'collector_returned': False}

    dp = SimpleNamespace(reject_conflicting_environment=lambda env: None,
                         RefmacDiagnosticProfile=lambda *args: object(),
                         FilePin=lambda **kwargs: kwargs, validate_profile=validate)
    col = SimpleNamespace(_resolve_ccp4_refmac_toolchain=lambda: ({}, {}, None),
                          run_refmac_rfactor_operation=collector)
    rerun = SimpleNamespace(_verify_no_protstruct_overrides=lambda: None)
    argv = [str(SOURCE), '--request', str(request_path), '--request-sha256', m.digest(request_path),
            '--request-size-bytes', str(request_path.stat().st_size)]
    original_path = sys.path[:]
    try:
        with patch.object(sys, 'argv', argv), \
             patch.object(sys, 'executable', str(Path(sys.executable).resolve())), \
             patch.object(m, 'safe_fresh_paths', return_value=None), \
             patch.object(m, 'isolate_environment', return_value={'synthetic': True}), \
             patch.object(m, 'validate_reviewed_git_state', return_value={'synthetic': True}), \
             patch.object(m, 'run_claimed_once', side_effect=claim) as mocked_claim, \
             patch.dict(os.environ, {'TMUX_PANE': '%0'}), \
             patch.dict(sys.modules, {'refmac_diagnostic_profile': dp, 'onesar_refmac': col,
                                      'onesar_rerun': rerun}), redirect_stdout(io.StringIO()):
            if reject:
                expect_refusal(name, m.main, 'Execution role mismatch')
                mocked_claim.assert_not_called()
                assert events == ['validate_profile']
            else:
                m.main()
                mocked_claim.assert_called_once()
                checks.append(name)
    finally:
        sys.path[:] = original_path
