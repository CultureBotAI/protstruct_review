def component_walk(raw):
    """Expand each symlink before processing its unconsumed suffix or '..'."""
    pending = deque(Path(raw).parts[1:])
    current = Path("/")
    edges = []
    steps = 0
    while pending:
        steps += 1
        assert steps < 10000, raw
        part = pending.popleft()
        if part == "..":
            current = current.parent
            continue
        if part == ".":
            continue
        node = current / part
        mode = node.lstat().st_mode
        if stat.S_ISLNK(mode):
            target = os.readlink(node)
            edges.append({"path": str(node), "target": target, "unconsumed_suffix": list(pending)})
            target_parts = Path(target).parts
            if target_parts and target_parts[0] == "/":
                current = Path("/")
                target_parts = target_parts[1:]
            pending.extendleft(reversed(target_parts))
        else:
            current = node
    return str(current), edges

def ancestry_prediction(raw):
    """Independent transcription of the inspected lines 71-83; not imported/executed composer."""
    pending = [Path(raw)]
    seen = set()
    edges = {}
    while pending:
        item = pending.pop()
        for node in (item, *item.parents):
            lexical = str(node)
            if lexical in seen:
                continue
            seen.add(lexical)
            if stat.S_ISLNK(node.lstat().st_mode):
                target = os.readlink(node)
                edges[lexical] = target
                pending.append(Path(target) if Path(target).is_absolute() else node.parent / target)
    return [{"path": k, "target": v,
             "physical_link_path": str(Path(k).parent.resolve(strict=True) / Path(k).name)}
            for k, v in sorted(edges.items())]

for raw, row in sorted(candidates.items()):
    try:
        row["leaf_lstat_is_symlink"] = stat.S_ISLNK(Path(raw).lstat().st_mode)
        resolved, edges = component_walk(raw)
        row["resolved_path"] = str(Path(raw).resolve(strict=True))
        assert resolved == row["resolved_path"], (raw, resolved, row["resolved_path"])
        row["resolved_lstat_is_regular"] = stat.S_ISREG(Path(resolved).lstat().st_mode)
        row["traversed_edges"] = edges
        row["ancestor_symlink"] = any(e["unconsumed_suffix"] for e in edges)
        row["predicted_composer_edges"] = ancestry_prediction(raw)
        retained = {(e["physical_link_path"], e["target"]) for e in row["predicted_composer_edges"]}
        row["unrepresented_traversed_edges"] = [e for e in edges if (e["path"], e["target"]) not in retained]
        row["retained_resolved_match"] = all(p == resolved for p in row["retained_resolved_paths"])
    except (OSError, AssertionError) as exc:
        row["error"] = f"{type(exc).__name__}: {exc}"
