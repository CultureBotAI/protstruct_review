def pin(path, target_key=None):
    path = Path(path)
    raw = path.read_bytes()
    result = {"path": str(path), "sha256": hashlib.sha256(raw).hexdigest(), "size_bytes": len(raw)}
    if target_key:
        result[target_key] = str(path.resolve(strict=True))
    return result


def dump(path, value):
    Path(path).write_text(json.dumps(value, indent=2) + "\n")


def make_fixture(directory):
    root = Path(directory).absolute()
    if root.exists() or root.is_symlink():
        raise ValueError("Fresh fixture directory required")
    if "601-instrumented-build" not in root.parts or "attempt04" in root.name.lower():
        raise ValueError("Synthetic orchestration fixture destination required")
    root.mkdir()
    trees = {name: root / name for name in ("vendor", "prepared", "build")}
    for path in trees.values():
        path.mkdir()
    files = {}
    roles = {}
    paths = {}

    def create(relative, content=None, executable=False):
        path = root / relative
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(content if content is not None else "inert fixture: " + relative + "\n")
        if executable:
            path.chmod(0o755)
        return path

    def register(path):
        path = Path(path)
        key = str(path)
        if key not in paths:
            ident = "item" + str(len(paths))
            paths[key] = ident
        ident = paths[key]
        edges = [{"path": str(q), "target": os.readlink(q)}
                 for q in [path, *path.parents] if q.is_symlink()]
        files[ident] = {**pin(path, "resolved_path"),
                        "symlinks": sorted(edges, key=lambda row: row["path"])}
        return ident

    def one(role, relative, content=None, executable=False):
        path = create(relative, content, executable)
        roles[role] = [register(path)]
        return path

    def inventory(tree):
        return {str(path.relative_to(tree)): {k: v for k, v in pin(path).items() if k != "path"}
                for path in sorted(tree.rglob("*")) if path.is_file()}

    original = create("vendor/source.f", "C inert upstream source\n")
    create("prepared/source.f", original.read_text())
    source_manifest = one("source_manifest", "prepared/refmac_diagnostic_source_manifest.json", "{}\n")
    patch = create("prepared/refmac_diagnostic_hooks.patch", "inert patch\n")
    writer = one("writer_input", "support/input/trace_writer.f90", "inert writer\n")
    hooks = one("hooks_input", "support/input/trace_hooks.f90", "inert hooks\n")
    create("prepared/trace_writer.f90", writer.read_text())
    create("prepared/trace_hooks.f90", hooks.read_text())
    for role, name in (
        ("preparation_receipt", "preparation-completed.json"),
        ("preparer", "prepare_refmac_diagnostic.py"),
        ("preparation_interpreter", "preparation-python"),
        ("source_auditor", "check-prepared-source.py"),
        ("source_audit_interpreter", "audit-python"),

        ("build_interpreter", "build-python"),
        ("build_qc", "focused-result.json"),
        ("build_qc_implementation", "focused-check.py"),

        ("launcher", "launch-attempt04.py"),
        ("collection_interpreter", "collection-python"),
    ):

                       ("build_runtime_config", "installed-runtime.cmake")):
        content = "inert pair: " + name + "\n"
        original_name = "build-instrumented-v2.py" if role == "build_driver" else name
        roles[role] = [register(create("support/original/" + original_name, content)),
                       register(create("build/" + name, content))]
    roles["build_inventories"] = []
    for name in ("toolchain-inventory.json", "effective-build-tools.json", "assembler-tool.json", "post-link-tools.json"):
        roles["build_inventories"].extend(register(create(prefix + name, "{}\n"))
                                          for prefix in ("support/original/", "build/"))
    roles["build_external_inputs"] = [register(create("support/compiler-input.txt"))]
    one("linked_executable", "build/refmac5.linked-unrelocated", "inert linked producer\n", True)
    executable = one("diagnostic_executable", "build/build/refmac5", "inert separate producer\n", True)
    create("build/ordinary-object.o", "inert object without an explicit role\n")

                          "upstream_source": {"expected_commit": COMMIT, "expected_tag": "5.8.0431"},
                          "extra_metadata_is_allowed": True})

    interpreter = Path(files[roles["source_audit_interpreter"][0]]["path"])

                 "source_root": str(trees["vendor"]), "prepared_root": str(trees["prepared"]),
                 "upstream_commit": COMMIT, "manifest_sha256": pin(source_manifest)["sha256"],
                 "auditor_sha256": pin(auditor)["sha256"], "patch_sha256": pin(patch)["sha256"],
                 "auditor_path": str(auditor), "interpreter": pin(interpreter, "resolved_path"),

    return {"manifest_path": manifest_path, "manifest": manifest,
            "installed_record": copy.deepcopy(installed), "environment": copy.deepcopy(env),
            "evidence_root": root / "collection-evidence-never-created",
