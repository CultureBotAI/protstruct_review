class ForbiddenQcOperation(RuntimeError):

def deny_processes(event, args):
    if event.startswith(("subprocess.", "os.exec", "os.spawn", "os.posix_spawn", "os.fork", "socket.")) or event == "os.system":
        denied_operations.append(event)
        raise ForbiddenQcOperation("Synthetic profile QC forbids process/network execution: " + event)
    write_paths = []
    if event == "open":
        path, mode, flags = args
        writing = isinstance(mode, str) and any(c in mode for c in "wax+")
        writing = writing or isinstance(flags, int) and bool(flags & (
            os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND))
        if writing:
            write_paths.append(path)
    elif event in {"os.mkdir", "os.remove", "os.rmdir", "os.chmod", "os.truncate"}:
        write_paths.append(args[0])
    elif event in {"os.rename", "os.link"}:
        write_paths.extend(args[:2])
    elif event == "os.symlink":
        write_paths.append(args[1])
    for path in write_paths:
        if not isinstance(path, int):
            resolved = Path(os.fsdecode(path)).resolve(strict=False)
            if resolved != out and not resolved.is_relative_to(out):
                denied_operations.append(event + ":outside-output")
                raise ForbiddenQcOperation("Synthetic QC write outside its fresh output: " + str(path))

def validate(f, request=None):
    installed_before = copy.deepcopy(f["installed_record"])
    environment_before = copy.deepcopy(f["environment"])
    supplied_request = request if request is not None else profile(f)
    f["held_request"] = supplied_request
    f["held_request_pins"] = {
        "manifest": supplied_request.manifest.to_dict(),
        "executable": supplied_request.executable.to_dict(),
    }
    f["held_installed_record"] = installed_before
    f["held_environment"] = environment_before
    f["held_outer_manifest"] = copy.deepcopy(f["manifest"])
    # Capture independent file pins before crossing the public API. Bad-shape
    # negative fixtures may contain unusable paths; they cannot become positives.
    f["held_file_pins"] = {}
    f["held_file_observations"] = {}
    for ident, row in f["manifest"]["files"].items():
        try:
            f["held_file_pins"][ident] = pin(row["path"], "resolved_path")
            f["held_file_observations"][ident] = capture_file_expectation(row["path"])
        except OSError as exc:
            f["held_file_pins"][ident] = {"unavailable": type(exc).__name__}
    f["held_manifest_observation"] = capture_file_expectation(f["manifest_path"])
    f["held_root_observations"] = {}
    for ident, row in f["manifest"]["roots"].items():
        try:
            f["held_root_observations"][ident] = {
                "path": row["path"], **path_identity(row["path"]),
                "members": expected_members(f, ident),
            }
        except OSError as exc:
            f["held_root_observations"][ident] = {"unavailable": type(exc).__name__}
    with ExitStack() as stack:
        if f.get("forbid_json_parse"):
            original_loads = json.loads
            forbidden = f["manifest_path"].read_bytes()
            f["manifest_parse_attempts"] = []
            f["referenced_inspection_attempts"] = []
            reference_paths = {
                os.path.normpath(row["path"]) for row in f["manifest"]["files"].values()
            } | {os.path.normpath(row["path"]) for row in f["manifest"]["roots"].values()}
            reference_paths.update(
                os.path.normpath(row["resolved_path"])
                for row in f["held_file_pins"].values() if "resolved_path" in row
            )
            def reference_trap(original):
                def checked(path, *args, **kwargs):
                    if isinstance(path, (str, bytes, os.PathLike)):
                        raw = os.fsdecode(path)
                        normalized = os.path.normpath(os.path.abspath(raw))
                        if normalized in reference_paths:
                            f["referenced_inspection_attempts"].append(raw)
                            raise AssertionError("Inspected a reference before manifest pin admission")

            stack.enter_context(patch.object(json, "loads", side_effect=only_manifest_trap))
            reference_functions = [(owner, name, getattr(owner, name)) for owner, name in (
                (builtins, "open"), (io, "open"), (os, "open"), (os, "stat"), (os, "lstat"),
                (os, "scandir"), (os, "listdir"))]

def path_identity(path, *, follow=False):
    value = os.stat(path, follow_symlinks=follow)
    return {"dev": value.st_dev, "ino": value.st_ino, "mode": stat.S_IMODE(value.st_mode),
            "uid": value.st_uid, "gid": value.st_gid}


def assert_identity(row, path, *, follow=False):
    assert isinstance(row, dict) and row["path"] == str(path)
    for key, value in path_identity(path, follow=follow).items():
        assert type(row[key]) is int and row[key] == value, (path, key)


def capture_file_expectation(path):
    """Independent fixture observation captured before the admission call."""
    raw = str(path)
    current = pin(raw, "resolved_path")
    parent_paths = {
        str(parent.resolve(strict=True))
        for spelling in (raw, current["resolved_path"])
        for parent in Path(spelling).parents
    }
    edge_paths = sorted(str(q) for q in (Path(raw), *Path(raw).parents) if q.is_symlink())
    return {
        **current, **path_identity(raw, follow=True),
        "ancestors": {p: {"path": p, **path_identity(p)} for p in parent_paths},
        "symlink_identities": {
            p: {"path": p, "target": os.readlink(p), **path_identity(p)} for p in edge_paths
        },
    }


def assert_held_file(observed, expected):
    for key, value in expected.items():
        if key in {"ancestors", "symlink_identities"}:
            indexed = {v["path"]: v for v in observed[key]}
            assert len(indexed) == len(observed[key])
            for path, identity in value.items():
                assert path in indexed, (key, path)
                for identity_key, identity_value in identity.items():
                    assert indexed[path][identity_key] == identity_value, (path, identity_key)
        else:
            assert observed[key] == value, (expected["path"], key)


def assert_ancestors(row, paths):
    observations = row["ancestors"]
    assert isinstance(observations, list) and observations
    indexed = {value["path"]: value for value in observations}
    assert len(indexed) == len(observations)
    # Enumerate before resolving: bin/../tool depends on bin as well as its
    # canonical parent, but the literal spelling bin/.. is not a directory ID.
    required = {str(parent.resolve(strict=True)) for path in paths for parent in Path(path).parents}
    assert required <= set(indexed), (required - set(indexed))
    for path, observation in indexed.items():
        assert_identity(observation, path)


def assert_file_observation(observation, path):
    raw_path = str(path)
    assert_identity(observation, raw_path, follow=True)
    current_pin = pin(raw_path, "resolved_path")
    assert type(observation["size_bytes"]) is int
    for key, value in current_pin.items():
        assert observation[key] == value, (raw_path, key)
    edges = observation["symlinks"]
    assert isinstance(edges, list)
    assert edges == sorted(edges, key=lambda edge: edge["path"])
    edge_paths = set()
    for edge in edges:
        assert isinstance(edge, dict) and isinstance(edge["path"], str)
        assert edge["target"] == os.readlink(edge["path"])
        edge_paths.add(edge["path"])
    assert len(edge_paths) == len(edges)
    # Independent fixture aliases have direct, deliberately known edges.
    direct_edges = {str(q) for q in (Path(raw_path), *Path(raw_path).parents) if q.is_symlink()}
    assert direct_edges <= edge_paths
    identities = observation["symlink_identities"]
    assert isinstance(identities, list)
    assert {v["path"] for v in identities} == edge_paths and len(identities) == len(edges)
    for identity in identities:
        assert_identity(identity, identity["path"])
    assert_ancestors(observation, (raw_path, current_pin["resolved_path"]))


def expected_members(f, root_id):
    root = Path(f["manifest"]["roots"][root_id]["path"])
    ids = {str(Path(row["path"]).relative_to(root)): ident for ident, row in f["manifest"]["files"].items()
           if Path(row["path"]).is_relative_to(root)}
    members = []
    def fail(exc):
        raise exc
    for current, directories, filenames in os.walk(root, followlinks=False, onerror=fail):
        if root_id == "vendor" and Path(current) == root:
            directories[:] = [n for n in directories if n != ".git"]
            filenames = [n for n in filenames if n != ".git"]
        for name in filenames:
            path = Path(current) / name
            relative = str(path.relative_to(root))
            members.append({"relative_path": relative, "file_id": ids.get(relative)})
    return sorted(members, key=lambda row: row["relative_path"])

def exception_chain_contains(cause, intended):
    seen = set()
    while isinstance(cause, Exception) and id(cause) not in seen:
        if cause is intended:
            return True
        seen.add(id(cause))
        cause = cause.__cause__
    return False

def reject_ordinary_field_writes(obj, names):
    """Try ordinary writes to actual existing evidence fields, without frozen-object bypasses."""
    checks = []
    for name in names:
        current = getattr(obj, name)
        for operation in ("assign", "delete"):
            try:
                if operation == "assign":
                    setattr(obj, name, current)
                else:
                    delattr(obj, name)
            except (AttributeError, TypeError):
                checks.append({"field": name, "operation": operation, "rejected": True})
            else:
                checks.append({"field": name, "operation": operation, "rejected": False})
                raise AssertionError("Writable retained evidence: " + name + "/" + operation)
    return checks


def reject_ordinary_item_writes(obj):
    assert len(obj), "Mutation control needs an existing item"
    key = next(iter(obj)) if isinstance(obj, Mapping) else 0
    value = obj[key]
    for operation in ("assign", "delete"):
        try:
            if operation == "assign":
                obj[key] = value
            else:
                del obj[key]
        except (AttributeError, TypeError):
            pass
        else:
            raise AssertionError("Writable retained container: " + operation)

def contaminate_return(value):
    if isinstance(value, dict):
        for key, item in list(value.items()):
            contaminate_return(item)
            if not isinstance(item, (dict, list, tuple)):
                value[key] = marker
        value[marker] = marker
    elif isinstance(value, (list, tuple)):
        for index, item in enumerate(list(value)):
            contaminate_return(item)
            if isinstance(value, list) and not isinstance(item, (dict, list, tuple)):
                value[index] = marker
        if isinstance(value, list):
            value.append(marker)

def capture_partial_dependencies(raw_path):
    """Capture measurable dependencies even when the final target is unavailable.

    This bounded independent fixture oracle uses lstat/readlink on each literal
    component before trying canonical directory resolution. It never substitutes
    the baseline identity for an unavailable live target.
    """
    path = Path(raw_path)
    parents, edges = {}, {}
    for candidate in (path, *path.parents):
        try:
            value = candidate.lstat()
        except OSError:
            continue
        if stat.S_ISLNK(value.st_mode):
            try:
                edges[str(candidate)] = {"path": str(candidate), "target": os.readlink(candidate),
                                         **path_identity(candidate)}
            except OSError:
                continue
        if candidate != path:
            try:
                canonical = str(candidate.resolve(strict=True))
                if stat.S_ISDIR(os.stat(canonical).st_mode):
                    parents[canonical] = {"path": canonical, **path_identity(canonical)}
            except OSError:
                continue
    return {"path": str(raw_path), "ancestors": parents, "symlink_identities": edges}


def capture_postflight_facts(f):
    """Observe the inert fixture independently before injected read/scan faults."""
    original = f["held_baseline_provenance"]
    facts = {}
    files = [("profile:manifest", original["manifest"])] + [
        ("profile:file:" + key, value) for key, value in original["files"].items()
    ]
    for key, row in files:
        try:
            facts[key] = {"complete": True, "file": capture_file_expectation(row["path"])}
        except OSError:
            facts[key] = {"complete": False, "file": capture_partial_dependencies(row["path"])}
    def fail(error):
        raise error
    for root_id, row in original["roots"].items():
        path = Path(row["path"])
        known_ids = {
            str(Path(value["path"]).relative_to(path)): ident
            for ident, value in original["files"].items()
            if Path(value["path"]).is_relative_to(path)
        }
        members = []
        subdirectories = {}
        for current, directories, filenames in os.walk(path, followlinks=False, onerror=fail):
            if root_id == "vendor" and Path(current) == path:
                directories[:] = [n for n in directories if n != ".git"]
                filenames = [n for n in filenames if n != ".git"]
            for name in directories:
                child = Path(current) / name
                subdirectories[str(child)] = {"path": str(child),
                    "resolved_path": str(child.resolve(strict=True)), **path_identity(child)}
            for name in filenames:
                relative = str((Path(current) / name).relative_to(path))
                members.append({"relative_path": relative, "file_id": known_ids.get(relative)})
        facts["profile:root:" + root_id] = {
            "path": str(path), "resolved_path": str(path.resolve(strict=True)),
            **path_identity(path), "members": sorted(members, key=lambda value: value["relative_path"]),
            "subdirectories": subdirectories,
            "ancestors": {
                str(parent): {"path": str(parent), **path_identity(parent)}
                for parent in (path, *path.parents)
            },
        }
    return facts


def assert_available_observation(key, row, fact):
    observed = row["observed"]
    if key.startswith("profile:root:"):
        # These fixtures fault enumeration, not the independently measured
        # root identity/dependencies. Errors cannot excuse dropping those facts.
        assert isinstance(observed, dict), (key, "missing obtainable root identity")
        for field in ("path", "resolved_path", "dev", "ino", "mode", "uid", "gid"):
            assert observed.get(field) == fact[field], (key, field)
        retained_ancestors = {value["path"]: value for value in observed.get("ancestors", [])}
        assert set(fact["ancestors"]) <= set(retained_ancestors), (key, "missing obtainable ancestors")
    if observed is None:
        assert row["errors"] and row["matches_baseline"] is False
        return
    assert isinstance(observed, dict) and observed["path"] == row["expected"]["path"]
    actual = fact if key.startswith("profile:root:") else fact["file"]
    for name in ("path", "resolved_path", "sha256", "size_bytes", "dev", "ino", "mode", "uid", "gid"):
        value = observed.get(name)
        if value is not None:
            assert name in actual and value == actual[name], (key, name, value, actual.get(name))
    for name in ("ancestors", "symlink_identities"):
        values = observed.get(name)
        if values is None:
            continue
        assert isinstance(values, list)
        assert len({value["path"] for value in values}) == len(values)
        for value in values:
            reference = actual.get(name, {}).get(value["path"])
            assert reference is not None, (key, name, value["path"])
            for field, expected_value in reference.items():
                assert value[field] == expected_value, (key, name, field)
    if not key.startswith("profile:root:"):
        edges = observed.get("symlinks")
        if edges is not None:
            expected_edges = [
                {"path": p, "target": value["target"]}
                for p, value in sorted(actual.get("symlink_identities", {}).items())
            ]
            assert edges == expected_edges
        if not row["errors"]:
            assert fact["complete"]
            assert_held_file(observed, actual)
    else:
        # Validate every supplied subdirectory identity, including partial rows.
        # Missing measurements may remain absent; false supplied identities may not.
        child_values = observed.get("subdirectories")
        children = {}
        if child_values is not None:
            assert isinstance(child_values, list)
            children = {value["path"]: value for value in child_values}
            assert len(children) == len(child_values)
            for child, value in children.items():
                assert child in actual["subdirectories"], (key, child)
                for field, reference in actual["subdirectories"][child].items():
                    assert value.get(field) == reference, (key, child, field)
        # A complete error-free observation must retain the independently known
        # identity. Calling omitted evidence a mismatch is not sufficient.
        if not row["errors"]:
            assert row["scan_complete"] is True
            for field in ("path", "resolved_path", "dev", "ino", "mode", "uid", "gid"):
                assert observed.get(field) == actual[field], (key, field)
            ancestors = {v["path"]: v for v in observed.get("ancestors", [])}
            assert set(actual["ancestors"]) <= set(ancestors), key
            assert child_values is not None
            assert set(children) == set(actual["subdirectories"]), key
        members = observed.get("members")
        if members is not None:
            known = {value["relative_path"]: value["file_id"] for value in actual["members"]}
            paths = [value["relative_path"] for value in members]
            assert paths == sorted(set(paths))
            for value in members:
                assert value["relative_path"] in known
                expected_id = known[value["relative_path"]]
                if expected_id is not None:
                    assert value["file_id"] == expected_id
                else:
                    assert isinstance(value["file_id"], str) and value["file_id"]
            if row["scan_complete"]:
                assert set(paths) == set(known)
        if row["scan_complete"]:
            assert members is not None


def descriptor_in_chain(descriptor, cause):
    keys = ("type", "message", "errno")
    if not isinstance(descriptor, dict) or any(key not in descriptor for key in keys):
        return False
    core = {key: descriptor[key] for key in keys}
    seen = set()
    while isinstance(cause, Exception) and id(cause) not in seen:
        if core == {"type": type(cause).__name__, "message": str(cause),
                    "errno": getattr(cause, "errno", None)}:
            return True
        seen.add(id(cause))
        cause = cause.__cause__
    return False


def assert_post_shape(f, baseline, observed, causes):
    original = f["held_baseline_provenance"]
    assert baseline.to_provenance() == original, "Postflight changed its original authority"
    assert observed["profile_id"] == original["profile_id"]
    assert set(observed["files"]) == set(original["files"])
    assert set(observed["roots"]) == {"vendor", "prepared", "build"}
    assert list(observed["files"]) == sorted(original["files"])
    assert list(observed["roots"]) == sorted(original["roots"])
    assert observed["runtime_image_coverage"] == "not_assessed"
    assert type(observed["verified"]) is bool
    assert len(causes) == len(observed["mutations"])
    assert isinstance(observed["errors"], list)
    keyed_rows = [("profile:manifest", observed["manifest"], original["manifest"])]
    keyed_rows += [("profile:file:" + k, r, original["files"][k]) for k, r in observed["files"].items()]
    keyed_rows += [("profile:root:" + k, r, original["roots"][k]) for k, r in observed["roots"].items()]
    # Fault keys come from independent fixture observations/injected objects,
    # never from the product's own error labels.
    observation_error_keys = set(f.get("injected_error_objects", {}))
    if f.get("injected_scan_exceptions"):
        observation_error_keys.add("profile:root:vendor")
    observation_error_keys.update(
        key for key, fact in f["postflight_facts"].items()
        if not key.startswith("profile:root:") and not fact["complete"]
    )
    for key, row, expected in keyed_rows:
        if key not in observation_error_keys:
            assert row["errors"] == [], (key, "invented observation error")
        assert row["expected"] == expected, key
        assert type(row["matches_baseline"]) is bool and isinstance(row["errors"], list)
        if key.startswith("profile:root:"):
            assert type(row["scan_complete"]) is bool
            assert isinstance(row["member_diff"], dict) and set(row["member_diff"]) == {"added", "removed"}
            if row["scan_complete"]:
                old_members = {v["relative_path"] for v in expected["members"]}
                new_members = {v["relative_path"] for v in row["observed"]["members"]}
                assert row["member_diff"] == {"added": sorted(new_members - old_members), "removed": sorted(old_members - new_members)}
            else:
                assert row["member_diff"]["removed"] == [], "Incomplete scan cannot establish missing members"
                seen = {value["relative_path"] for value in (row["observed"] or {}).get("members", [])}
                old = {value["relative_path"] for value in expected["members"]}
                assert set(row["member_diff"]["added"]) <= seen - old
        assert_available_observation(key, row, f["postflight_facts"][key])
        complete = not key.startswith("profile:root:") or row["scan_complete"]
        identity_fields = ("path", "resolved_path", "dev", "ino", "mode", "uid", "gid", "ancestors")
        if key.startswith("profile:root:"):
            identity_fields += ("name", "members", "subdirectories")
        else:
            identity_fields += ("sha256", "size_bytes", "symlinks", "symlink_identities")
        same_identity = row["observed"] is not None and all(
            row["observed"].get(field) == expected.get(field) for field in identity_fields
        )
        independent_match = complete and not row["errors"] and same_identity
        assert row["matches_baseline"] is bool(independent_match), key
        if not independent_match:
            assert any(m["key"] == key for m in observed["mutations"]), key
        for error in row["errors"]:
            assert_error_descriptor(error)
            assert error.get("key", key) == key, (key, "wrong row error owner")
            assert any(e["key"] == key and all(e.get(n) == error[n] for n in ("type", "message", "errno")) for e in observed["errors"])
            assert any(m["key"] == key and m["error"] == error for m in observed["mutations"])
    for mutation, cause in zip(observed["mutations"], causes):
        assert set(mutation) >= {"category", "key", "path", "reason", "expected", "observed", "error"}
        assert isinstance(mutation["reason"], str) and mutation["reason"]
        indexed_rows = {key: row for key, row, _ in keyed_rows}
        assert mutation["key"] in indexed_rows
        matched_row = indexed_rows[mutation["key"]]
        assert matched_row["matches_baseline"] is False
        assert mutation["path"] == matched_row["expected"]["path"]
        assert mutation["expected"] == matched_row["expected"]
        assert mutation["observed"] == matched_row["observed"]
        category = "diagnostic_profile_" + mutation["key"].split(":")[1]
        assert mutation["category"] == category
        if mutation["key"] not in observation_error_keys:
            assert mutation["error"] is None and cause is None, (mutation["key"], "comparison-only cause")
        if mutation["error"] is None:
            assert cause is None
        else:
            assert_error_descriptor(mutation["error"])
            assert isinstance(cause, Exception)
            assert descriptor_in_chain(mutation["error"], cause)
            assert mutation["error"] in matched_row["errors"]
            injected = list(f.get("injected_scan_exceptions", [])) if mutation["key"] == "profile:root:vendor" else []
            if mutation["key"] in f.get("injected_error_objects", {}):
                injected.append(f["injected_error_objects"][mutation["key"]])
            if injected:
                assert any(exception_chain_contains(cause, original_error)
                           for original_error in injected), (mutation["key"], "reconstructed cause")
    for error in observed["errors"]:
        indexed_rows = {key: row for key, row, _ in keyed_rows}
        assert error["key"] in indexed_rows
        assert any(row_error.get("key", error["key"]) == error["key"]
                   and all(row_error.get(field) == error[field]
                           for field in ("type", "message", "errno"))
                   for row_error in indexed_rows[error["key"]]["errors"])
    independent_verified = all(row["matches_baseline"] for _, row, _ in keyed_rows)
    independent_verified = independent_verified and not observed["mutations"] and not observed["errors"]
    assert observed["verified"] is bool(independent_verified)
    if f.get("injected_scan_exceptions"):
        assert any(m["key"] == "profile:root:vendor"
                   and any(exception_chain_contains(cause, original_error)
                           for original_error in f["injected_scan_exceptions"])
                   for m, cause in zip(observed["mutations"], causes))
    for key, original_error in f.get("injected_error_objects", {}).items():
        assert any(m["key"] == key and exception_chain_contains(cause, original_error)
                   for m, cause in zip(observed["mutations"], causes)), key

def case(name, mutate=None, expected_reject=True, post=None, post_assert=None, post_context=None,
         expected_check=None, mutation_descriptor=None, pre_context=None, failure_assert=None):
    f = make_fixture(out / ("fixture-" + str(len(results))))
    observed = None
    if pre_context is not None:
        f["admission_context"] = pre_context
    phase = "mutation"
    denied_before = len(denied_operations)
    declared_check = expected_check or PREFLIGHT_EXPECTATIONS.get(name)
    if expected_reject and not post and declared_check is None:
        raise ValueError("Missing independently declared failure stage for " + name)
    try:
        if post:
            baseline = validate(f)
            f["held_baseline_provenance"] = copy.deepcopy(assert_admitted(f, baseline))
            post(f)
            f["postflight_facts"] = capture_postflight_facts(f)
            phase = "postflight"
            causes = []
            with ExitStack() as stack:
                if post_context is not None:
                    stack.enter_context(post_context(f))
                observed = module.recheck_profile(baseline, mutation_causes_out=causes)
            assert capture_postflight_facts(f) == f["postflight_facts"], "Recheck changed the inert fixture"
            json.dumps(observed, allow_nan=False)
            assert_post_shape(f, baseline, observed, causes)
            passed = (observed["verified"] is (not expected_reject)
                      and post_assert is not None and post_assert(f, baseline, observed))
        else:
            request = mutate(f) if mutate else None
            phase = "validation"
            observed = validate(f, request)
            passed = not expected_reject
            if passed:
                assert_admitted(f, observed)
    except Exception as exc:
        intended_type = getattr(module, "ProfileValidationError", None)
        intended = intended_type is not None and type(exc) is intended_type
        passed = expected_reject and not post and phase == "validation" and intended
        returned_observation = copy.deepcopy(observed) if isinstance(observed, dict) else None
        observed = {"error_type": type(exc).__name__, "message": str(exc),
                    "traceback": traceback.format_exc()}
        if returned_observation is not None:
            observed["returned_observation"] = returned_observation
        if passed:
            try:
                failure = exc.to_provenance()
                json.dumps(failure, allow_nan=False)
                assert failure["profile_files_verified"] is False
                assert failure["selected_executable"] is None
                assert failure["runtime_image_coverage"] == "not_assessed"
                assert failure["failed_check"] == declared_check, (name, declared_check, failure["failed_check"])
                assert failure["requested"] == f["held_request_pins"]
                assert_error_descriptor(failure["error"])
                if f.get("forbid_json_parse"):
                    assert f["manifest_parse_attempts"] == []
                    assert f["referenced_inspection_attempts"] == []
                observed["failed_admission"] = failure
                if failure_assert is not None:
                    failure_assert(f, exc, failure)
                if name == "repinned source_audit.passed":
                    observed["failure_immutability_checks"] = assert_failure_immutable(exc)
            except Exception as check_error:
                passed = False
                observed["failure_assertion_error"] = type(check_error).__name__ + ": " + str(check_error)
                observed["failure_assertion_traceback"] = traceback.format_exc()
    passed = passed and len(denied_operations) == denied_before
    row = {"name": name, "passed": passed, "expect_rejection": expected_reject,
           "expected_check": declared_check, "mutation_descriptor": mutation_descriptor,
           "caller_request_pins": f.get("held_request_pins"),
           "postflight": post is not None, "phase": phase,
           "cause_identity_checks": f.get("retained_exception_identity_checks"),
           "observation": observed if isinstance(observed, dict) else type(observed).__name__}
    results.append(row)
    dump(out / "cases.json", results)
    return passed

def same_byte_replace(path):
    raw = path.read_bytes()
    mode = path.stat().st_mode
    path.rename(path.with_name(path.name + ".saved-original"))
    path.write_bytes(raw)
    path.chmod(mode)


def change_bytes_same_size(path):
    original = path.read_bytes()
    assert original, 'Same-size corruption needs a nonempty inert file'
    changed = bytes([original[0] ^ 1]) + original[1:]
    assert len(changed) == len(original) and changed != original
    path.write_bytes(changed)
    assert path.read_bytes() == changed


def changed_role(f, observed, role):
    ident = f["manifest"]["roles"][role][0]
    row = observed["files"][ident]
    return (row["matches_baseline"] is False
            and any(m["key"] == "profile:file:" + ident for m in observed["mutations"]))


def all_rows_readable(observed):
    return (observed["manifest"]["observed"] is not None
            and all(r["observed"] is not None for r in observed["files"].values())
            and all(r["observed"] is not None and r["scan_complete"] is True
                    for r in observed["roots"].values()))


def replace_directory_preserving_children(target, backup):
    mode = stat.S_IMODE(target.stat().st_mode)
    before = {str(path.relative_to(target)): path_identity(path)
              for path in target.rglob("*") if path.is_file()}
    target.rename(backup)
    target.mkdir()
    target.chmod(mode)
    for child in backup.iterdir():
        child.rename(target / child.name)
    after = {str(path.relative_to(target)): path_identity(path)
             for path in target.rglob("*") if path.is_file()}
    assert before == after, "Directory replacement changed child file identities"


def alias_row(f, observed, index):

    ident = next(k for k, value in f["manifest"]["files"].items() if value["path"] == path)

            and all(value["matches_baseline"] for key, value in observed["files"].items() if key != ident)
            and all(value["matches_baseline"] for value in observed["roots"].values())
            and observed["manifest"]["matches_baseline"])

@contextmanager
def root_scan_failure(f):
    target = f["root"] / "vendor"
    original_scandir, original_listdir = os.scandir, os.listdir
    f["injected_scan_failures"] = []
    f["injected_scan_exceptions"] = []
    def wrap(original):
        def fail_target(path=".", *args, **kwargs):
            if not isinstance(path, int) and Path(path).absolute() == target:
                f["injected_scan_failures"].append(str(path))
                error = PermissionError(errno.EACCES, "deliberate controlled-root scan failure", str(path))
                f["injected_scan_exception"] = error
                f["injected_scan_exceptions"].append(error)
                raise error
            return original(path, *args, **kwargs)
        return fail_target
    with ExitStack() as stack:
        # Cover both common traversal primitives and direct imports of them.
        for name, value in list(vars(module).items()):
            if value is original_scandir:
                stack.enter_context(patch.object(module, name, wrap(original_scandir)))
            elif value is original_listdir:
                stack.enter_context(patch.object(module, name, wrap(original_listdir)))
        stack.enter_context(patch.object(os, "scandir", wrap(original_scandir)))
        stack.enter_context(patch.object(os, "listdir", wrap(original_listdir)))
        yield


@contextmanager
def partial_scan_and_read_failure(f):
    """Prepared fault fixture: a child scan fails after root members were read,
    and an external file read fails after descriptor identity was measured.
    Partial-result assertions check the accepted failure representation below.
    """
    scan_target = f["root"] / "build/build"
    read_target = role_path(f, "launcher")
    identity = read_target.stat()
    read_key = "profile:file:" + f["manifest"]["roles"]["launcher"][0]
    scan_error = PermissionError(errno.EACCES, "deliberate child-directory scan failure", str(scan_target))
    read_error = OSError(errno.EIO, "deliberate post-fstat read failure", str(read_target))
    f["injected_error_objects"] = {read_key: read_error, "profile:root:build": scan_error}
    f["partial_fault_events"] = []
    original_scandir, original_listdir = os.scandir, os.listdir
    original_fdopen, original_fstat = os.fdopen, os.fstat
    build_root = f["root"] / "build"
    root_member_identities = {}
    for child in build_root.iterdir():
        if child.is_file():
            st = child.stat()
            root_member_identities[(st.st_dev, st.st_ino)] = child.name
    scan_state = {"root_started": False, "child_failed": False}
    f["partial_completed_root_members"] = []
    def tracked_fstat(fd):
        result = original_fstat(fd)
        if (result.st_dev, result.st_ino) == (identity.st_dev, identity.st_ino):
            f["partial_fault_events"].append({"stage": "consumer_fstat", "key": read_key,
                "dev": result.st_dev, "ino": result.st_ino, "mode": stat.S_IMODE(result.st_mode),
                "uid": result.st_uid, "gid": result.st_gid, "stat_size_bytes": result.st_size})
        return result
    class FailedRead:
        def __init__(self, handle):
            self.handle = handle
            self.prefix_read = False
        def __enter__(self):
            self.handle.__enter__()
            return self
        def __exit__(self, *args):
            return self.handle.__exit__(*args)
        def fileno(self):
            return self.handle.fileno()
        def read(self, size=-1):
            prefix = f.get("partial_read_prefix_bytes", 0)
            if prefix and not self.prefix_read:
                self.prefix_read = True
                value = self.handle.read(min(size, prefix) if size >= 0 else prefix)
                f["partial_fault_events"].append({"stage": "successful_read", "key": read_key,
                                                  "size_bytes": len(value)})
                return value
            f["partial_fault_events"].append({"stage": "read_failure", "key": read_key})
            raise read_error
    class ObservedRootRead:
        def __init__(self, handle, member):
            self.handle, self.member = handle, member
            self.reached_eof = False
            self.completed_before_child_failure = False
            self.byte_count = 0
        def __enter__(self):
            self.handle.__enter__()
            return self
        def __exit__(self, exc_type, exc, tb):
            if exc_type is None and self.completed_before_child_failure:
                f["partial_completed_root_members"].append(self.member)
                f["partial_fault_events"].append({"stage": "completed_root_member_stream",
                    "key": "profile:root:build", "relative_path": self.member,
                    "size_bytes": self.byte_count})
            return self.handle.__exit__(exc_type, exc, tb)
        def fileno(self):
            return self.handle.fileno()
        def read(self, size=-1):
            value = self.handle.read(size)
            self.byte_count += len(value)
            if size < 0 or (value == b"" and size != 0):
                self.reached_eof = True
                # Completion time and successful closure are separate facts.
                # A later close must not discard an earlier completed read.
                if not scan_state["child_failed"]:
                    self.completed_before_child_failure = True
            return value
    def fdopen(fd, *args, **kwargs):
        st = original_fstat(fd)
        handle = original_fdopen(fd, *args, **kwargs)
        key = (st.st_dev, st.st_ino)
        if key == (identity.st_dev, identity.st_ino):
            return FailedRead(handle)
        if scan_state["root_started"] and not scan_state["child_failed"] and key in root_member_identities:
            return ObservedRootRead(handle, root_member_identities[key])
        return handle
    def scan(original):
        def wrapped(path=".", *args, **kwargs):
            if not isinstance(path, int) and Path(path).absolute() == build_root:
                scan_state["root_started"] = True
            if not isinstance(path, int) and Path(path).absolute() == scan_target:
                scan_state["child_failed"] = True
                f["partial_fault_events"].append({"stage": "child_scan_failure", "key": "profile:root:build"})
                raise scan_error
            return original(path, *args, **kwargs)
        return wrapped
    with ExitStack() as stack:
        for owner, name, replacement in (
            (os, "fstat", tracked_fstat), (os, "fdopen", fdopen),
            (os, "scandir", scan(original_scandir)), (os, "listdir", scan(original_listdir)),
        ):
            original = getattr(owner, name)
            for imported_name, value in list(vars(module).items()):
                if value is original:
                    stack.enter_context(patch.object(module, imported_name, replacement))
            stack.enter_context(patch.object(owner, name, replacement))
        yield


def assert_partial_file_read(f, observed, expected):
    """Require actual completed measurements, with unknown full hash/size."""
    ident = f["manifest"]["roles"]["launcher"][0]
    key = "profile:file:" + ident
    assert isinstance(observed, dict)
    for field in ("path", "resolved_path", "dev", "ino", "mode", "uid", "gid"):
        assert observed[field] == expected[field], field
    assert observed["stat_size_bytes"] == expected["size_bytes"]
    assert observed["sha256"] is None and observed["size_bytes"] is None
    assert observed["read_complete"] is False
    events = [e for e in f["partial_fault_events"] if e["key"] == key]
    names = [e["stage"] for e in events]
    assert "consumer_fstat" in names and "read_failure" in names
    assert names.index("consumer_fstat") < names.index("read_failure")
    for event in events:
        if event["stage"] == "consumer_fstat":
            for field in ("dev", "ino", "mode", "uid", "gid"):
                assert observed[field] == event[field]
    streamed = sum(e["size_bytes"] for e in events if e["stage"] == "successful_read")
    assert observed["streamed_size_bytes"] == streamed
    prefix = f.get("partial_read_prefix_bytes", 0)
    assert 0 < streamed <= prefix if prefix else streamed == 0

def inspect_immutable(value, path="baseline", visited=None):
